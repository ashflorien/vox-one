cmake_minimum_required(VERSION 3.22)
project(VoxOne VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(APPLE)
  set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "" FORCE)
endif()
include(FetchContent)
FetchContent_Declare(juce GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(juce)
juce_add_plugin(VoxOne COMPANY_NAME "VoxOne Audio" BUNDLE_ID "audio.voxone.plugin" PLUGIN_MANUFACTURER_CODE VoxA PLUGIN_CODE VxO1 FORMATS AU VST3 PRODUCT_NAME "VoxOne" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
juce_generate_juce_header(VoxOne)
target_sources(VoxOne PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(VoxOne PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(VoxOne PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_gui_extra juce::juce_recommended_config_flags juce::juce_recommended_lto_flags juce::juce_recommended_warning_flags)
